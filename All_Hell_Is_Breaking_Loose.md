# First, download [http://www.doomworld.com/idgames/index.php?id=6480 allhell.zip] containing the game files, and extract it.
# Run with the following options:
 chocolate-doom -merge allhell.wad ahflats.wad -file ahlevels.wad ahmus.wad -deh allhell.deh
